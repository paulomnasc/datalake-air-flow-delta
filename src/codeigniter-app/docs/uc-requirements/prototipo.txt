import React, { useState, useEffect, useRef } from 'react';
import { 
  Play, 
  CheckCircle, 
  ChevronRight, 
  Terminal, 
  BookOpen, 
  Award, 
  Zap,
  Layout,
  Database,
  Code
} from 'lucide-react';

const App = () => {
  const [sidebarOpen, setSidebarOpen] = useState(true);
  const [activeTask, setActiveTask] = useState(0);
  const [progress, setProgress] = useState(35);
  const [points, setPoints] = useState(450);
  const [isCompleted, setIsCompleted] = useState(false);
  const syncStateRef = useRef({ lastAt: 0, timeoutId: null });

  const STORAGE_KEY = 'uc-progress-monitor-state';
  const API_URL = '/api/uc-progress';
  const userContext = {
    userId: 'ID_DO_USER_LOGADO',
    courseId: 'curso-001',
    moduleId: 'mod-006'
  };

  // Simulação de tarefas práticas
  const tasks = [
    {
      id: 0,
      title: "Configurar Fonte S3",
      description: "Conecte o bucket de staging para iniciar o Lake.",
      videoTimestamp: "02:15",
      points: 100
    },
    {
      id: 1,
      title: "Validar Schema",
      description: "Execute o comando de validação no console ao lado.",
      videoTimestamp: "05:40",
      points: 150
    },
    {
      id: 2,
      title: "Deploy da Camada Silver",
      description: "Finalize a transformação dos dados brutos.",
      videoTimestamp: "12:10",
      points: 200
    }
  ];

  useEffect(() => {
    const saved = localStorage.getItem(STORAGE_KEY);
    if (!saved) {
      return;
    }

    try {
      const parsed = JSON.parse(saved);
      if (typeof parsed.activeTask === 'number') {
        setActiveTask(parsed.activeTask);
      }
      if (typeof parsed.progress === 'number') {
        setProgress(parsed.progress);
      }
      if (typeof parsed.points === 'number') {
        setPoints(parsed.points);
      }
      if (typeof parsed.isCompleted === 'boolean') {
        setIsCompleted(parsed.isCompleted);
      }
    } catch (error) {
      localStorage.removeItem(STORAGE_KEY);
    }
  }, []);

  useEffect(() => {
    const completedTaskIds = tasks
      .filter((_, idx) => idx < activeTask)
      .map(task => task.id);

    const payload = {
      ...userContext,
      activeTask,
      progress,
      points,
      isCompleted,
      completedTaskIds,
      timestamp: new Date().toISOString()
    };

    localStorage.setItem(STORAGE_KEY, JSON.stringify(payload));

    const now = Date.now();
    const lastAt = syncStateRef.current.lastAt;
    const elapsed = now - lastAt;
    const delay = elapsed > 2000 ? 0 : 500;

    if (syncStateRef.current.timeoutId) {
      clearTimeout(syncStateRef.current.timeoutId);
    }

    syncStateRef.current.timeoutId = setTimeout(() => {
      syncStateRef.current.lastAt = Date.now();
      fetch(API_URL, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify(payload)
      }).catch(() => {});
    }, delay);
  }, [activeTask, progress, points, isCompleted, tasks]);

  const completeTask = (id) => {
    if (id === activeTask) {
      const nextTask = activeTask + 1;
      setPoints(prev => prev + tasks[id].points);
      if (nextTask < tasks.length) {
        setActiveTask(nextTask);
        setProgress(((nextTask) / tasks.length) * 100);
      } else {
        setProgress(100);
        setIsCompleted(true);
      }
    }
  };

  return (
    <div className="flex h-screen bg-slate-950 text-slate-100 font-sans overflow-hidden">
      
      {/* 1. ÁREA DE EXECUÇÃO PRÁTICA (O seu "Módulo de Trabalho") */}
      <main className={`flex-1 flex flex-col transition-all duration-300 ${sidebarOpen ? 'mr-0' : ''}`}>
        {/* Header do Sistema de Trabalho */}
        <header className="h-14 border-b border-slate-800 bg-slate-900/50 flex items-center justify-between px-6">
          <div className="flex items-center gap-3">
            <div className="bg-indigo-600 p-1.5 rounded-lg">
              <Database size={20} />
            </div>
            <span className="font-bold tracking-tight text-slate-200">DATAFLOW <span className="text-indigo-400 text-xs">PRO</span></span>
          </div>
          
          <div className="flex items-center gap-6">
            <div className="flex items-center gap-2 px-3 py-1 bg-amber-500/10 border border-amber-500/20 rounded-full">
              <Zap size={14} className="text-amber-500 fill-amber-500" />
              <span className="text-sm font-bold text-amber-500">{points} XP</span>
            </div>
            <button 
              onClick={() => setSidebarOpen(!sidebarOpen)}
              className="p-2 hover:bg-slate-800 rounded-lg text-slate-400"
            >
              <Layout size={20} />
            </button>
          </div>
        </header>

        {/* Workspace Central */}
        <div className="flex-1 p-6 overflow-auto">
          <div className="max-w-4xl mx-auto space-y-6">
            <div className="flex justify-between items-end">
              <div>
                <h1 className="text-2xl font-bold">Workspace: Pipeline Alpha</h1>
                <p className="text-slate-400">Ambiente de laboratório para engenharia de dados</p>
              </div>
              <div className="text-xs text-slate-500 bg-slate-900 px-3 py-1 rounded border border-slate-800">
                STATUS: AGUARDANDO AÇÃO
              </div>
            </div>

            {/* Simulação de um Terminal/Console de Trabalho */}
            <div className="bg-black rounded-xl border border-slate-800 overflow-hidden shadow-2xl">
              <div className="bg-slate-900 px-4 py-2 border-b border-slate-800 flex items-center gap-2">
                <Terminal size={14} className="text-slate-500" />
                <span className="text-xs font-mono text-slate-400">bash — data-engine-v1</span>
              </div>
              <div className="p-6 font-mono text-sm space-y-2 h-[400px]">
                <p className="text-green-500">$ data-lake init --region us-east-1</p>
                <p className="text-slate-500">Initializing project structure...</p>
                <p className="text-slate-300">[OK] Repository created.</p>
                <p className="text-indigo-400">$ data-flow connect --source s3://staging-bucket</p>
                <p className="text-slate-300">Attempting connection to S3...</p>
                {activeTask > 0 && <p className="text-green-500">[SUCCESS] Connection established with 42 objects.</p>}
                
                {/* Botão de "Execução Real" que gatilha progresso no treinamento */}
                <div className="mt-8 pt-4 border-t border-slate-800">
                  <button 
                    onClick={() => completeTask(activeTask)}
                    className="flex items-center gap-2 bg-indigo-600 hover:bg-indigo-500 text-white px-6 py-2 rounded-lg font-semibold transition-colors shadow-lg shadow-indigo-500/20"
                  >
                    <Play size={16} fill="white" />
                    Executar Comando da Tarefa {activeTask + 1}
                  </button>
                </div>
              </div>
            </div>

            <div className="grid grid-cols-3 gap-4">
              {[1, 2, 3].map(i => (
                <div key={i} className="h-32 bg-slate-900/50 border border-slate-800 rounded-xl border-dashed flex items-center justify-center text-slate-600 italic text-sm">
                  Espaço para Logs/Métricas
                </div>
              ))}
            </div>
          </div>
        </div>
      </main>

      {/* 2. BARRA LATERAL DE TREINAMENTO (O "Fosso" Eliminado) */}
      <aside 
        className={`${sidebarOpen ? 'w-96' : 'w-0'} bg-slate-900 border-l border-slate-800 transition-all duration-300 flex flex-col relative`}
      >
        <div className="p-6 border-b border-slate-800">
          <div className="flex items-center gap-2 text-indigo-400 mb-2">
            <BookOpen size={16} />
            <span className="text-xs font-bold uppercase tracking-widest">Guia de Aprendizado</span>
          </div>
          <h2 className="text-lg font-bold leading-tight">Módulo 6: Criando um Data Lake do Zero</h2>
          
          <div className="mt-4 space-y-1">
            <div className="flex justify-between text-xs mb-1">
              <span className="text-slate-400">Progresso do Módulo</span>
              <span className="text-indigo-400 font-bold">{Math.round(progress)}%</span>
            </div>
            <div className="h-2 bg-slate-800 rounded-full overflow-hidden">
              <div 
                className="h-full bg-indigo-500 transition-all duration-700"
                style={{ width: `${progress}%` }}
              ></div>
            </div>
          </div>
        </div>

        <div className="flex-1 overflow-y-auto p-6 space-y-8">
          {/* Player de Vídeo Compacto (Floating) */}
          <div className="aspect-video bg-black rounded-lg border border-slate-700 relative group overflow-hidden shadow-xl">
             <div className="absolute inset-0 flex items-center justify-center bg-slate-800/50 group-hover:bg-slate-800/30 transition-all">
                <Play size={32} className="text-white fill-white opacity-80" />
             </div>
             <div className="absolute bottom-0 left-0 right-0 h-1 bg-slate-700">
                <div className="h-full bg-red-600 w-1/3"></div>
             </div>
             <div className="absolute top-2 right-2 bg-black/60 px-2 py-1 rounded text-[10px] font-mono">
                {tasks[activeTask]?.videoTimestamp || "FIM"}
             </div>
          </div>

          {/* Checklist de Tasks Interativas */}
          <div className="space-y-4">
            <h3 className="text-sm font-semibold text-slate-400 flex items-center gap-2">
              <CheckCircle size={14} /> TAREFAS DE EXECUÇÃO
            </h3>
            
            {tasks.map((task, idx) => (
              <div 
                key={task.id}
                className={`p-4 rounded-xl border transition-all ${
                  activeTask === idx 
                    ? 'bg-indigo-500/10 border-indigo-500/50 shadow-[0_0_15px_rgba(99,102,241,0.1)]' 
                    : idx < activeTask 
                    ? 'bg-slate-800/30 border-slate-800 opacity-60' 
                    : 'bg-slate-900 border-slate-800 opacity-40'
                }`}
              >
                <div className="flex justify-between items-start mb-2">
                  <span className={`text-[10px] font-bold px-2 py-0.5 rounded ${
                    activeTask === idx ? 'bg-indigo-500 text-white' : 'bg-slate-700 text-slate-300'
                  }`}>
                    PASSO {idx + 1}
                  </span>
                  {idx < activeTask && <CheckCircle size={16} className="text-green-500" />}
                </div>
                <h4 className={`font-bold text-sm ${activeTask === idx ? 'text-white' : 'text-slate-400'}`}>
                  {task.title}
                </h4>
                <p className="text-xs text-slate-500 mt-1">{task.description}</p>
                
                {activeTask === idx && (
                  <div className="mt-3 pt-3 border-t border-indigo-500/20 flex items-center justify-between">
                    <span className="text-[10px] text-indigo-400 font-mono italic">Assista até {task.videoTimestamp}</span>
                    <span className="text-[10px] font-bold text-amber-500">+{task.points} XP</span>
                  </div>
                )}
              </div>
            ))}
          </div>

          {isCompleted && (
            <div className="bg-gradient-to-br from-indigo-600 to-purple-600 p-6 rounded-xl text-center shadow-lg shadow-indigo-500/20">
              <Award size={40} className="mx-auto mb-2 text-amber-300" />
              <h3 className="font-bold text-lg leading-tight">Módulo Concluído!</h3>
              <p className="text-xs opacity-90 mb-4">Você dominou a fundação do Data Lake.</p>
              <button className="w-full bg-white text-indigo-600 py-2 rounded-lg font-bold text-sm">
                Emitir Certificado
              </button>
            </div>
          )}
        </div>

        {/* Footer da barra de treinamento */}
        <div className="p-4 bg-slate-950 border-t border-slate-800">
           <button className="w-full flex items-center justify-center gap-2 text-xs text-slate-500 hover:text-white transition-colors">
              <span>Dúvidas com o Wallace?</span>
              <ChevronRight size={14} />
           </button>
        </div>
      </aside>
    </div>
  );
};

export default App;
